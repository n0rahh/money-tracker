# Money Tracker — Milestone 2 (Insights) Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Answer "was this month more expensive than last, and why?" in two taps: a Trends tab with monthly bars over a chosen range, a month-vs-month category comparison, a per-category detail page, and on Home a pace line, per-category deltas, a daily strip and one-tap suggestion chips in the Add sheet.

**Architecture:** Everything is computed on the client from Firestore range queries already supported by the one composite index (`date DESC, at DESC`): Home subscribes to the current and the previous month, Trends to the displayed range plus one month before it (for deltas), the category page to twelve months, the Add sheet to the last 60 days. New pure functions in `src/lib/stats.ts` do the arithmetic (tested with TDD); charts are hand-written HTML/CSS bar columns (accessible buttons, no chart library). Navigation gains a Trends tab and a `/category/:id` route; Home accepts `?month=YYYY-MM` so Trends can deep-link into it.

**Tech Stack:** unchanged from M1 — React 19, TypeScript strict, Vite 8, Tailwind v4, Firebase 13, react-router 7, Vitest 5 + Testing Library, lucide-react.

**Spec:** `docs/superpowers/specs/2026-10-14-money-tracker-design.md` — §7 (Home, Add sheet suggestion chips, Trends, Category detail), §9 (analytics definitions, binding), §10 "M2 — Insights" row, §11.

## Global Constraints

- All M1 constraints hold (see `docs/superpowers/plans/2026-10-14-m1-core-logging.md` → Global Constraints): integer minor units, local ISO dates, 44 px targets, never-await writes with `.catch` → toast, Inter + `tnum`, pristine test output, conventional commits ending with `Co-Authored-By: Claude Fable 5.1 <noreply@anthropic.com>`, no new dependencies (no chart library).
- Analytics definitions are spec §9, verbatim: month = calendar month local time; spent/income = sums by kind; net = income − spent; share = category ÷ kind total; **pace** = spent from day 1 through today vs the previous month over the same number of days, delta in % with a sign, only for the current month; **Δ vs last month** = per-category difference of full-month totals, absolute and %, "new" when the previous total is 0, −100 % when the current is 0; **monthly trend** = one bar per month over the range, income as dots, range inclusive of the current month; **compare A vs B** = per-category table of two months sorted by |Δ|; **suggestions** = group the last 60 days by (kind, categoryId, amount, note), rank by count then recency, take 4, exclude combos already used today.
- Ranges: 3M / 6M / 12M = the current month and the 2 / 5 / 11 months before it; YTD = January of the current year through the current month.
- Data windows: Home subscribes to the current month and the previous month; Trends to `[rangeStart − 1 month, end of current month]`; the category page to the last 12 months; the Add sheet to the last 60 days (which also feeds category recency).
- Production JS ≤ 350 kB gzipped (M1 ships at ~295 kB; this milestone must add < 30 kB).
- Deep links: `/?month=YYYY-MM` opens Home on that month (invalid values are ignored); `/category/:id` with an unknown id shows "Category not found" and a link back.

```ts
/** Inclusive ascending list of months; empty when `from` is after `to`. */
export function monthsBetween(from: IsoMonth, to: IsoMonth): IsoMonth[] {
  const out: IsoMonth[] = []
  for (let m = from; m <= to; m = addMonths(m, 1)) out.push(m)
  return out
}

export function yearToDateMonths(now: Date = new Date()): IsoMonth[] {
  return monthsBetween(`${now.getFullYear()}-01`, currentMonth(now))
}

/** "Oct" in the current year, "Oct 25" otherwise. */
export function monthShortLabel(month: IsoMonth, locale?: string | null, now: Date = new Date()): string {
  const [y, m] = month.split('-').map(Number)
  const label = new Intl.DateTimeFormat(locale ?? undefined, { month: 'short' }).format(new Date(y, m - 1, 1))
  return y === now.getFullYear() ? label : `${label} ${String(y).slice(-2)}`
}
```

```ts
export type RangeKey = '3M' | '6M' | '12M' | 'YTD'

export function rangeMonths(range: RangeKey, now: Date = new Date()): IsoMonth[] {
  if (range === 'YTD') return yearToDateMonths(now)
  const count = range === '3M' ? 3 : range === '6M' ? 6 : 12
  return lastMonths(currentMonth(now), count)
}

export type MonthTotals = { month: IsoMonth; expense: number; income: number; net: number }

export function monthlyTotals(txs: Transaction[], months: IsoMonth[]): MonthTotals[] {
  const index = new Map(months.map((m, i) => [m, i]))
  const rows = months.map((month) => ({ month, expense: 0, income: 0, net: 0 }))
  for (const t of txs) {
    const i = index.get(monthOf(t.date))
    if (i === undefined) continue
    if (t.kind === 'expense') rows[i].expense += t.amount
    else rows[i].income += t.amount
  }
  for (const r of rows) r.net = r.income - r.expense
  return rows
}

export type Pace = { spentToDate: number; previousToDate: number; dayOfMonth: number; deltaPct: number | null }

const pct = (current: number, previous: number): number | null =>
  previous === 0 ? null : Math.round(((current - previous) / previous) * 100)

export function paceComparison(current: Transaction[], previous: Transaction[], today: IsoDate): Pace {
  const dayOfMonth = Number(today.slice(8, 10))
  const previousMonth = addMonths(monthOf(today), -1)
  const cutoff = Math.min(dayOfMonth, daysInMonth(previousMonth))
  let spentToDate = 0
  for (const t of current) if (t.kind === 'expense' && t.date <= today) spentToDate += t.amount
  let previousToDate = 0
  for (const t of previous) {
    if (t.kind !== 'expense' || monthOf(t.date) !== previousMonth) continue
    if (Number(t.date.slice(8, 10)) <= cutoff) previousToDate += t.amount
  }
  return { spentToDate, previousToDate, dayOfMonth, deltaPct: pct(spentToDate, previousToDate) }
}

export type CategoryDelta = { categoryId: string; current: number; previous: number; delta: number; deltaPct: number | null }

export function categoryDeltas(current: Transaction[], previous: Transaction[], kind: Kind): CategoryDelta[] {
  const sum = (txs: Transaction[]) => {
    const m = new Map<string, number>()
    for (const t of txs) if (t.kind === kind) m.set(t.categoryId, (m.get(t.categoryId) ?? 0) + t.amount)
    return m
  }
  const a = sum(current)
  const b = sum(previous)
  const ids = new Set([...a.keys(), ...b.keys()])
  return [...ids]
    .map((categoryId) => {
      const cur = a.get(categoryId) ?? 0
      const prev = b.get(categoryId) ?? 0
      return { categoryId, current: cur, previous: prev, delta: cur - prev, deltaPct: cur === 0 && prev > 0 ? -100 : pct(cur, prev) }
    })
    .sort((x, y) => Math.abs(y.delta) - Math.abs(x.delta) || x.categoryId.localeCompare(y.categoryId))
}

export type Suggestion = { kind: Kind; categoryId: string; amount: number; note: string; count: number; lastAt: number }

export function suggestions(txs: Transaction[], today: IsoDate, limit = 4): Suggestion[] {
  const groups = new Map<string, Suggestion & { usedToday: boolean }>()
  for (const t of txs) {
    const key = `${t.kind}|${t.categoryId}|${t.amount}|${t.note}`
    const g = groups.get(key) ?? { kind: t.kind, categoryId: t.categoryId, amount: t.amount, note: t.note, count: 0, lastAt: 0, usedToday: false }
    g.count += 1
    g.lastAt = Math.max(g.lastAt, t.at.toMillis())
    if (t.date === today) g.usedToday = true
    groups.set(key, g)
  }
  return [...groups.values()]
    .filter((g) => !g.usedToday)
    .sort((x, y) => y.count - x.count || y.lastAt - x.lastAt)
    .slice(0, limit)
    .map(({ usedToday: _usedToday, ...s }) => s)
}

export function categoryMonthlyTotals(txs: Transaction[], categoryId: string, months: IsoMonth[]): number[] {
  const index = new Map(months.map((m, i) => [m, i]))
  const out = months.map(() => 0)
  for (const t of txs) {
    if (t.categoryId !== categoryId) continue
    const i = index.get(monthOf(t.date))
    if (i !== undefined) out[i] += t.amount
  }
  return out
}
```
